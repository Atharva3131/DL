import numpy as np
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import mean_absolute_error,mean_squared_error
from tensorflow.keras.models import Model
from tensorflow.keras.layers import Input,LSTM,Dense

t=np.arange(1000)
s=10+.01*t+np.sin(2*np.pi*t/50)+.1*np.random.randn(1000)
scaler=StandardScaler(); s=scaler.fit_transform(s[:,None]).flatten()

X,Y=[],[]
for i in range(len(s)-40):
    X.append(s[i:i+30]); Y.append(s[i+30:i+40])
X,Y=np.array(X)[...,None],np.array(Y)[...,None]
n=int(.8*len(X))

a=Input((30,1)); _,h,c=LSTM(64,return_state=True)(a)
b=Input((10,1)); x=LSTM(64,return_sequences=True)(b,[h,c])
out=Dense(1)(x)

model=Model([a,b],out); model.compile('adam','mse',metrics=['mae'])
model.fit([X[:n],np.zeros_like(Y[:n])],Y[:n],epochs=20,batch_size=32,verbose=0)

p=model.predict([X[n:],np.zeros_like(Y[n:])],verbose=0)
p=scaler.inverse_transform(p.reshape(-1,1))
actual=scaler.inverse_transform(Y[n:].reshape(-1,1))
print("MAE:",round(mean_absolute_error(actual,p),4))
print("RMSE:",round(np.sqrt(mean_squared_error(actual,p)),4))
